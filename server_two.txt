const express = require('express');
const fs = require('fs/promises');
const path = require('path');

const app = express();
const PORT = 3000;
const DATA_FILE = path.join(__dirname, 'data.json');

app.use(express.json());

app.use((req, res, next) => {
    const timestamp = new Date().toLocaleTimeString();
    console.log(`[${timestamp}] ${req.method} ${req.url}`);
    next();
})

// middleware function to validate note data
function validateNote(req, res, next){
    const {title, content} = req.body;
    if(!title || !title.trim() || !content || !content.trim()){
        return res.status(400).
        json({error: 'Title and content are required'});
    }

    next();
}
// GET endpoint to retrieve data from the JSON file
async function getNotesFromFile(){
    try{
        const data = await fs.readFile(DATA_FILE, 'utf-8');
        return JSON.parse(data);
    }catch(error){
        return [];
    }

}

async function saveNotesToFile(notes){
        await fs.writeFile(DATA_FILE, JSON.stringify(notes, null, 2), 'utf-8');
    
}

app.get('/notes', async (req, res) =>{
    try{
        const notes = await getNotesFromFile();
        res.json(notes);
    }catch(error){
        res.status(500).json({ error: 'Failed to retrieve notes' });
    }
})

// သတ်မှတ်ထားသော id ဖြင့် note ရှာရန် GET endpoint
app.get('/notes/:id', async (req, res) =>{
    const notes = await getNotesFromFile();
    const noteId = parseInt(req.params.id);
    const note = notes.find(n => n.id === noteId);

    try{
        if(!note){
            res.status(404).json({message: 'Note not found'});

        }
        res.json(note);
    }catch(error){
        res.status(500).json({error: 'Failed to retrieve note'});

    }
});

app.post('/notes',validateNote, async (req, res) =>{
    try{
        const {title, content} = req.body;

        if(!title || !content){
            return res.status(400).json({error: 'Title and content are required'});

        }
        const notes = await getNotesFromFile();

        const newNote = {
            id: notes.length > 0 ? notes[notes.length - 1].id + 1 : 1,
            title,
            content
        };

        notes.push(newNote);
        await saveNotesToFile(notes);

        res.status(201).json(newNote);


    }
    catch(error){
        res.status(500).json({message: 'Failed to create note'});
    }
})

// put endpoint to update a note by id
app.put('/notes/:id',validateNote, async (req, res) =>{
    try{
        const noteId = parseInt(req.params.id);
        const notes = await getNotesFromFile();
        const note = notes.find(n => n.id === noteId);
        
        if(!note){
            return res.status(404).json({message: 'Note not found'});
        }
        
        const {title, content} = req.body;
        if(title) note.title = title;
        if(content) note.content = content;
        await saveNotesToFile(notes);
        res.json(note);
       
    

    }catch(error){
        res.status(500).json({message: 'Failed to update note'});
    }
});

app.delete('/notes/:id', async (req, res) =>{
    try{
        const noteId = parseInt(req.params.id);
        const notes = await getNotesFromFile();
        const noteIndex = notes.findIndex(n => n.id === noteId);

        if(noteIndex === -1){
            return res.status(404).json({message: 'Note not found'});
        }
        notes.splice(noteIndex, 1);
        await saveNotesToFile(notes);
        res.json({message: 'Note deleted successfully'});


    }catch(error){
        res.status(500).json({message: 'Failed to delete note'});
    }
})

app.listen(PORT, () =>{
    console.log(`Server is running on http://localhost:${PORT}`);
});
